#define BLYNK_PRINT Serial
#define BLYNK_TEMPLATE_ID ""			//Blynk Daten eintragen
#define BLYNK_TEMPLATE_NAME ""
#define BLYNK_AUTH_TOKEN  ""

#include <ESP8266WiFi.h>
#include <WiFiUdp.h>
#include <BlynkSimpleEsp8266.h>



//LED PINS
#define LEDR_PIN D1
#define LEDY_PIN D2
#define LED_WIFI D3

//BUZZER PINS
#define BUZZER1_PIN D5
#define BUZZER2_PIN D6

//TASTER PIN
#define TASTER_PIN D7

//WLAN & UDP
const char* ssid = "";				//WLAN Daten hier eintragen
const char* password = "";			

//BLYNK
char auth[] = BLYNK_AUTH_TOKEN;


WiFiUDP udp;
const unsigned int localPort = 4210;


//EMPFANGSVARIABLEN
bool alarmEmpfangen = false,  alarmAn = false;

//LED VARIABLEN
bool ledStatus = false;
unsigned long toggleZuletzt = 0, toggleInterval = 200;
unsigned long blinkZuletzt = 0, blinkInterval = 500;

//TASTER VARIABLEN
bool tasterZuletzt = HIGH, tasterJetzt = HIGH;

//--------FUNKTIONEN-----------

//UDP NACHRICHTEN EMPFANGEN
void pruefeUDP() {
 int packetSize = udp.parsePacket();
 if (packetSize) {
   char buffer[255];
   int len = udp.read(buffer, 254);
   if (len > 0) buffer[len] = 0; // Null-terminieren
   String msg = String(buffer);
   msg.trim();

   if (msg == "ALARM_ON") {
     alarmEmpfangen = true;
     Serial.println("ALARM vom Sender empfangen!");
   } else if (msg == "ALARM_OFF") {
     alarmEmpfangen = false;
     alarmAn = false;
     Serial.println("ALARM vom Sender deaktiviert!");
   } else {
     Serial.println("Unbekannte Nachricht: " + msg);
   }
 }
}

//FUNKTIONEN ALARM

void pruefeAlarm(){
 if(alarmEmpfangen){
   alarmAn= true;
 }
}

//FUNKTIONEN TASTER
// ---- FUNKTION: Alarm manuell einschalten ----
void tasterSetztAlarm() {
 tasterJetzt = digitalRead(TASTER_PIN);

//FLANKENABFRAGE
 if (tasterZuletzt == HIGH && tasterJetzt == LOW) {
   alarmAn = !alarmAn;       // Alarm umschalten
   alarmEmpfangen = false;    // Sender-Signal ignorieren
   Serial.println(alarmAn ? "Alarm manuell aktiviert" : "Alarm manuell deaktiviert");
 }
 tasterZuletzt = tasterJetzt;
}

BLYNK_WRITE(V0) {
  int value = param.asInt();

  if (value == 1) {
    alarmAn = true;
    alarmEmpfangen = false;
    Serial.println("Alarm über Blynk EIN");
  } else {
    alarmAn = false;
    alarmEmpfangen = false;
    Serial.println("Alarm über Blynk AUS");
  }
}

void updateAlarm(){
 if(alarmAn){
//LED&BUZZER TOGGLE
   if (millis() - toggleZuletzt >= toggleInterval) {
     ledStatus = !ledStatus;
     digitalWrite(LEDR_PIN, ledStatus ? HIGH : LOW);
     digitalWrite(BUZZER2_PIN, ledStatus ? HIGH : LOW);
     digitalWrite(LEDY_PIN, !ledStatus ? HIGH : LOW);
     digitalWrite(BUZZER1_PIN, ledStatus ? HIGH : LOW);
     toggleZuletzt = millis();
 }
} else {
 
   digitalWrite(BUZZER1_PIN, LOW);
   digitalWrite(BUZZER2_PIN, LOW);
   digitalWrite(LEDR_PIN, LOW);
   digitalWrite(LEDY_PIN, LOW);
 }
}
void updateLedWLAN() {
  if (WiFi.status() == WL_CONNECTED) {
    digitalWrite(LED_WIFI, HIGH);
  }else digitalWrite(LED_WIFI, LOW);
}

void setup() {
// put your setup code here, to run once:
 Serial.begin(9600);

 pinMode(LED_WIFI, OUTPUT);
 pinMode(LEDR_PIN, OUTPUT);
 pinMode(LEDY_PIN, OUTPUT);
 pinMode(BUZZER1_PIN, OUTPUT);
 pinMode(BUZZER2_PIN, OUTPUT);
 pinMode(TASTER_PIN, INPUT_PULLUP);

 digitalWrite(LED_WIFI, LOW);
 digitalWrite(LEDR_PIN, LOW);
 digitalWrite(LEDY_PIN, LOW);
 digitalWrite(BUZZER1_PIN, LOW);
 digitalWrite(BUZZER2_PIN, LOW);


//WLAN verbinden
// WLAN verbinden
 WiFi.begin(ssid, password);
 Serial.print("Verbinde mit WLAN");

//UDP starten
 udp.begin(localPort);
 Serial.println("UDP-Empfang gestartet auf Port " + String(localPort));


//BLYNK INITIALISIEREN
Blynk.begin(auth, ssid, password);
}
void loop() {
// put your main code here, to run repeatedly:

if (WiFi.status() != WL_CONNECTED) {
   WiFi.reconnect();
 }

updateLedWLAN();
Blynk.run();
pruefeUDP();
tasterSetztAlarm();
pruefeAlarm();
updateAlarm();

Blynk.virtualWrite(V1, alarmAn ? 255 : 0);

}
